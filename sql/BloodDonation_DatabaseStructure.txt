###############################################################################
# Database Structure: Blood Donation Management
#
# **CORRECT ORDER FOR FOREIGN KEY DEPENDENCIES**
###############################################################################

CREATE DATABASE donor_portal;
USE donor_portal;

-- 1. BASE / REFERENCE TABLES (No outgoing foreign keys)

-- TABLE: spender (Donor)
-- Contains information about registered blood donors.
-------------------------------------------------------------------------------
CREATE TABLE spender (
    donor_id VARCHAR(15) PRIMARY KEY,       -- Unique ID of the donor
    last_donation_date DATE,               -- Date of the last blood donation
    INDEX (donor_id)                       -- Index for quick search of donors
);

-- TABLE: ort (Location)
-- Contains details about the locations where blood donation appointments take place.
-------------------------------------------------------------------------------
CREATE TABLE location (
    location_id VARCHAR(15) PRIMARY KEY,   -- Unique ID of the location
    name VARCHAR(50),                      -- Name of the location
    street VARCHAR(50),                    -- Street of the location
    city VARCHAR(20),                      -- City of the location
    postal_code INT,                       -- Postal code of the location
    INDEX (name),                          -- Index for searching by location name
    INDEX (city),                          -- Index for searching by city
    INDEX (postal_code)                    -- Index for searching by postal code
);

-- TABLE: status (Appointment Status)
-- Defines the possible status values for blood donation appointments.
-------------------------------------------------------------------------------
CREATE TABLE status (
    status_id INT PRIMARY KEY,             -- Unique ID of the status
    status_desc VARCHAR(30),               -- Description of the status (e.g., Planned, Cancelled)
    INDEX (status_desc)                    -- Index for searching by status description
);

-- TABLE: faq_sections (FAQ Sections)
-- Categorizes frequently asked questions into different sections.
-------------------------------------------------------------------------------
CREATE TABLE faq_sections (
    id INT PRIMARY KEY,                    -- Unique ID of the section
    section_name VARCHAR(50),              -- Name of the section
    INDEX (section_name)                   -- Index for searching by section name
);

-- TABLE: admins (Administrators)
-- Contains information about database administrators.
-------------------------------------------------------------------------------
CREATE TABLE admins (
    admin_id INT PRIMARY KEY,              -- Unique ID of the administrator
    username VARCHAR(50) UNIQUE,           -- Username of the administrator (unique)
    password VARCHAR(255),                 -- Encrypted password of the administrator
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- Timestamp of account creation
    INDEX (username)                       -- Index for searching by username
);


---

-- 2. MEDIUM DEPENDENCY TABLES

-- TABLE: spendetermine (Appointments)
-- Manages information about planned and past blood donation appointments.
-- DEPENDS ON: location, status
-------------------------------------------------------------------------------
CREATE TABLE appointments (
    appointment_id INT PRIMARY KEY,        -- Unique ID of the appointment
    location_id VARCHAR(15),               -- Foreign key to location
    status_id INT,                         -- Foreign key to status
    date DATE,                             -- Date of the appointment
    time_start TIME,                       -- Start time of the appointment
    time_end TIME,                         -- End time of the appointment
    appointment_group VARCHAR(50),         -- Group assignment (e.g., blood type)
    appointment_slot VARCHAR(50),          -- Designation of the slot
    FOREIGN KEY (location_id) REFERENCES location(location_id),
    FOREIGN KEY (status_id) REFERENCES status(status_id),
    INDEX (location_id),                   -- Index for searching appointments at a location
    INDEX (status_id),                     -- Index for searching appointments by status
    INDEX (date, time_start)               -- Index for time-based searches
);


---

-- 3. HIGH DEPENDENCY TABLES

-- TABLE: spende (Donation)
-- Records details of individual blood donation events.
-- DEPENDS ON: spender, appointments
-------------------------------------------------------------------------------
CREATE TABLE donation (
    donation_id INT PRIMARY KEY,           -- Unique ID of the donation
    donor_id VARCHAR(15),                  -- Foreign key to spender (Donor)
    appointment_id INT,                    -- Foreign key to appointments
    FOREIGN KEY (donor_id) REFERENCES spender(donor_id),
    FOREIGN KEY (appointment_id) REFERENCES appointments(appointment_id),
    INDEX (donor_id),                      -- Index for searching donations by donor
    INDEX (appointment_id)                 -- Index for searching donations by appointment
);

-- TABLE: chat_faq (FAQ Entries)
-- Stores frequently asked questions and their answers.
-- DEPENDS ON: faq_sections
-------------------------------------------------------------------------------
CREATE TABLE chat_faq (
    id INT PRIMARY KEY,                    -- Unique ID of the FAQ entry
    section_id INT,                        -- Foreign key to faq_sections (Section)
    question VARCHAR(255),                 -- The question
    answer TEXT,                           -- The answer
    FOREIGN KEY (section_id) REFERENCES faq_sections(id),
    INDEX (section_id),                    -- Index for searching questions within a section
    FULLTEXT (question, answer)            -- Full-text index for keyword search
);


-- TABLE: admin_permission (Admin Permissions)
-- Links administrators with their specific permissions.
-- DEPENDS ON: admins
-------------------------------------------------------------------------------
CREATE TABLE admin_permission (
    admin_id INT,                          -- Foreign key to admins (Administrator)
    permission_name VARCHAR(50),           -- Assigned permission
    FOREIGN KEY (admin_id) REFERENCES admins(admin_id),
    INDEX (admin_id),                      -- Index for searching permissions of an admin
    INDEX (permission_name)                -- Index for searching by permission
);
